# 1. Clone the repo
git clone https://github.com/prem-programs/Socrates.git
cd Socrates

# 2. Install Node.js dependencies
npm install

# 3. Compile TypeScript → JavaScript
npm run compile

# 4. Install Python backend dependencies
pip install -r backend/requirements.txt
